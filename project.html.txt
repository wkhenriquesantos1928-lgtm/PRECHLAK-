  const renderFormattedText = (text) => {
    if (!text) return "";
    
    const parts = text.split(/(```[\s\S]*?```)/g);

    return parts.map((part, index) => {
      if (part.startsWith('```') && part.endsWith('```')) {
        const codeContent = part.replace(/```[a-zA-Z]*\n?/, '').replace(/```$/, '');
        return (
          <div key={index} className="my-3 font-mono text-xs bg-slate-950 p-3 rounded-lg border border-slate-800 overflow-x-auto text-emerald-400 select-all">
            <pre className="whitespace-pre">{codeContent}</pre>
          </div>
        );
      }

      const lines = part.split('\n');
      return (
        <div key={index} className="space-y-2">
          {lines.map((line, lIdx) => {
            if (line.startsWith('###')) {
              const titleText = line.replace('###', '').trim();
              let iconElement = <FileText className="w-4 h-4 inline mr-2" />;
              
              if (titleText.includes("1.") || titleText.includes("RESUMO")) iconElement = <BookOpen className="w-4 h-4 inline mr-2" />;
              if (titleText.includes("2.") || titleText.includes("ANÁLISE")) iconElement = <Search className="w-4 h-4 inline mr-2" />;
              if (titleText.includes("3.") || titleText.includes("RISCOS")) iconElement = <AlertTriangle className="w-4 h-4 inline mr-2" />;
              if (titleText.includes("4.") || titleText.includes("CONCLUSÃO")) iconElement = <Shield className="w-4 h-4 inline mr-2" />;
              if (titleText.includes("5.") || titleText.includes("RECOMENDAÇÕES")) iconElement = <CheckCircle2 className="w-4 h-4 inline mr-2" />;

              return (
                <h4 key={lIdx} className="text-sm font-bold tracking-wider border-b border-slate-800/80 pb-1 mt-4 mb-2 flex items-center uppercase font-mono">
                  {iconElement}
                  {titleText}
                </h4>
              );
            }

            if (line.startsWith('*') || line.startsWith('-')) {
              const content = line.substring(1).trim();
              return (
                <div key={lIdx} className="flex items-start gap-2 pl-3 text-slate-300 text-sm">
                  <span className="mt-1.5 h-1 w-1 rounded-full bg-slate-400 flex-shrink-0"></span>
                  <p>{replaceBold(content)}</p>
                </div>
              );
            }

            if (!line.trim()) return <div key={lIdx} className="h-2"></div>;

            return (
              <p key={lIdx} className="text-sm text-slate-300 leading-relaxed pl-1 font-mono">
                {replaceBold(line)}
              </p>
            );
          })}
        </div>
      );
    });
  };

  const replaceBold = (text) => {
    const parts = text.split(/(\*\*.*?\*\*)/g);
    return parts.map((part, i) => {
      if (part.startsWith('**') && part.endsWith('**')) {
        const rawText = part.slice(2, -2);
        const isSuspect = rawText.toUpperCase().includes('SUSPEITO') || rawText.toUpperCase().includes('ALTO RISCO') || rawText.toUpperCase().includes('DANGER') || rawText.toUpperCase().includes('GOLPE') || rawText.toUpperCase().includes('FAKE');
        
        return (
          <strong 
            key={i} 
            className={`font-bold uppercase tracking-wider ${
              isSuspect 
                ? 'text-red-500 bg-red-900/40 px-1.5 py-0.5 rounded border border-red-800 animate-pulse' 
                : 'text-slate-100'
            }`}
          >
            {rawText}
          </strong>
        );
      }
      return part;
    });
  };

  const getStatusConfig = (status) => {
    if (isMonochrome) {
      switch (status) {
        case 'ALTO RISCO':
        case 'SUSPEITO':
          return {
            bg: 'bg-slate-900',
            border: 'border-white',
            badgeBg: 'bg-white text-slate-950 border-white',
            glow: 'shadow-none',
            icon: AlertTriangle
          };
        case 'SEGURO':
        default:
          return {
            bg: 'bg-slate-950',
            border: 'border-slate-800',
            badgeBg: 'bg-slate-900 text-slate-400 border-slate-800',
            glow: 'shadow-none',
            icon: CheckCircle2
          };
      }
    } else {
      switch (status) {
        case 'ALTO RISCO':
        case 'SUSPEITO':
          return {
            bg: 'bg-red-900/10 border-red-900/60',
            border: 'border-red-600',
            badgeBg: 'bg-red-950/70 text-red-400 border-red-800',
            glow: 'shadow-[0_0_15px_rgba(220,38,38,0.15)]',
            icon: AlertTriangle
          };
        case 'SEGURO':
        default:
          return {
            bg: 'bg-emerald-950/20',
            border: 'border-emerald-500',
            badgeBg: 'bg-emerald-950/60 text-emerald-300 border-emerald-600',
            glow: 'shadow-[0_0_15px_rgba(16,185,129,0.15)]',
            icon: CheckCircle2
          };
      }
    }
  };

  const statusStyle = getStatusConfig(currentRiskStatus);

  // Mapeamento de estilos baseados no modo monocromático
  const borderAccent = isMonochrome ? 'border-slate-850' : 'border-slate-800/80';
  const textAccent = isMonochrome ? 'text-slate-200' : 'text-cyan-400';
  const buttonPrimary = isMonochrome 
    ? 'bg-slate-100 hover:bg-white text-slate-950' 
    : 'bg-gradient-to-r from-cyan-600 to-blue-600 hover:from-cyan-500 hover:to-blue-500 text-white';

  // --- SE NÃO ESTIVER LOGADO, EXIBE APENAS TELA DE LOGIN ---
  if (!isLoggedIn) {
    return (
      <div className="min-h-screen bg-slate-950 text-slate-100 flex items-center justify-center p-4">
        <div className="max-w-md w-full bg-slate-900 border border-slate-800 rounded-2xl p-6 shadow-2xl relative overflow-hidden">
          
          <div className="absolute top-0 right-0 w-32 h-32 bg-cyan-500/5 rounded-full blur-3xl"></div>
          <div className="absolute bottom-0 left-0 w-32 h-32 bg-red-500/5 rounded-full blur-3xl"></div>

          <div className="flex flex-col items-center gap-3 mb-6">
            <div className="p-3 bg-gradient-to-tr from-cyan-600 to-blue-700 rounded-xl border border-cyan-500/30">
              <Shield className="w-8 h-8 text-white animate-pulse" />
            </div>
            <div className="text-center">
              <h2 className="text-xl font-black tracking-widest font-mono uppercase">
                PRECHLAK<span className="text-cyan-400">OSINT</span>
              </h2>
              <p className="text-xs text-slate-400 font-mono tracking-tight mt-1">
                SISTEMA OPERACIONAL DE INTELIGÊNCIA
              </p>
            </div>
          </div>

          <form onSubmit={handleLogin} className="space-y-4 font-mono">
            {loginError && (
              <div className="p-3 bg-red-950/40 border border-red-900/60 rounded-lg text-red-400 text-xs flex items-center gap-2">
                <AlertTriangle className="w-4 h-4 shrink-0" />
                <span>{loginError}</span>
              </div>
            )}

            <div className="space-y-1">
              <label className="text-[10px] font-bold text-slate-400 uppercase">Usuário:</label>
              <input 
                type="text"
                value={loginUsername}
                onChange={(e) => setLoginUsername(e.target.value)}
                className="w-full bg-slate-950 border border-slate-800 focus:border-cyan-500/50 rounded-lg px-3 py-2 text-sm text-slate-100 focus:outline-none"
                placeholder="Insira sua credencial"
                required
              />
            </div>

            <div className="space-y-1">
              <label className="text-[10px] font-bold text-slate-400 uppercase">Senha de Segurança:</label>
              <input 
                type="password"
                value={loginPassword}
                onChange={(e) => setLoginPassword(e.target.value)}
                className="w-full bg-slate-950 border border-slate-800 focus:border-cyan-500/50 rounded-lg px-3 py-2 text-sm text-slate-100 focus:outline-none"
                placeholder="••••••••"
                required
              />
            </div>

            <button
              type="submit"
              className="w-full py-2.5 bg-gradient-to-r from-cyan-600 to-blue-600 hover:from-cyan-500 hover:to-blue-500 text-white font-bold rounded-lg text-xs transition-all shadow-md flex items-center justify-center gap-2 mt-2"
            >
              <Lock className="w-4 h-4" />
              <span>Autenticar Operador</span>
            </button>
          </form>

          <div className="mt-6 pt-4 border-t border-slate-800 text-[10px] text-center text-slate-500 font-mono">
            Canal de auditoria regulamentado de segurança corporativa.
          </div>
        </div>
      </div>
    );
  }

  return (
    <div className={`min-h-screen bg-slate-950 text-slate-100 font-sans flex flex-col antialiased ${isMonochrome ? 'selection:bg-white/20' : 'selection:bg-cyan-500/30'}`}>
      
      {/* NOTIFICAÇÃO DO SISTEMA */}
      {alertMsg && (
        <div className={`border-b text-xs px-4 py-2 text-center font-mono flex items-center justify-center gap-2 animate-pulse sticky top-0 z-50 ${isMonochrome ? 'bg-slate-900 border-slate-750 text-slate-200' : 'bg-red-950 border-red-850 text-red-400'}`}>
          <Activity className={`w-4 h-4 animate-spin ${isMonochrome ? 'text-white' : 'text-red-400'}`} />
          <span>{alertMsg}</span>
        </div>
      )}

      {/* HEADER OPERACIONAL */}
      <header className="bg-slate-900/90 border-b border-slate-800/80 sticky top-0 z-40 backdrop-blur-md px-4 py-3">
        <div className="max-w-7xl mx-auto flex flex-col md:flex-row justify-between items-center gap-4">
          
          <div className="flex items-center gap-3">
            <div className={`p-2 rounded-lg shadow-lg border ${isMonochrome ? 'bg-slate-800 border-slate-700' : 'bg-gradient-to-tr from-cyan-600 to-blue-700 border-cyan-500/30 shadow-cyan-500/10'}`}>
              <Shield className="w-6 h-6 text-white animate-pulse" />
            </div>
            <div>
              <div className="flex items-center gap-2">
                <h1 className="text-lg font-black tracking-widest text-slate-100 uppercase font-mono">
                  PRECHLAK<span className={isMonochrome ? 'text-white' : 'text-cyan-400'}>OSINT</span>
                </h1>
                <span className={`text-[10px] px-1.5 py-0.5 rounded border font-mono font-bold ${isMonochrome ? 'bg-slate-800 text-slate-300 border-slate-700' : 'bg-red-950/40 text-red-400 border-red-900'}`}>
                  v7.0 PRO
                </span>
              </div>
              <p className="text-xs text-slate-400 font-mono tracking-tight flex items-center gap-1.5">
                <span className={`w-1.5 h-1.5 rounded-full animate-ping inline-block ${isMonochrome ? 'bg-white' : 'bg-red-500'}`}></span>
                SUÍTE DE AUDITORIA FORENSE E CONTROLE DE ACESSO
              </p>
            </div>
          </div>

          <div className="flex items-center gap-3 w-full md:w-auto justify-end">
            
            {/* BOTÃO ALTERNADOR DE TEMA */}
            <button
              onClick={() => {
                setIsMonochrome(!isMonochrome);
                triggerAlert(`Tema alterado para: ${!isMonochrome ? 'Tactical Noir' : 'Cyber Neon'}`);
              }}
              className={`px-3 py-1.5 rounded-lg border text-xs font-mono font-bold transition-all flex items-center gap-1.5 ${
                isMonochrome 
                  ? 'bg-slate-800 border-slate-700 text-white' 
                  : 'bg-slate-950 border-slate-800 text-slate-400 hover:text-white'
              }`}
            >
              {isMonochrome ? <EyeOff className="w-4 h-4" /> : <Eye className="w-4 h-4" />}
              <span>{isMonochrome ? 'MONO: ATIVO' : 'TEMA: CYBER'}</span>
            </button>

            <div className={`px-3 py-1.5 rounded border ${statusStyle.badgeBg} flex items-center gap-2 text-xs font-mono tracking-wider uppercase transition-all duration-300 ${statusStyle.glow}`}>
              <statusStyle.icon className="w-4 h-4 animate-bounce" />
              <span>SISTEMA: <strong className="font-bold">{currentRiskStatus}</strong></span>
            </div>

            <button 
              onClick={() => setShowSettings(!showSettings)}
              className={`p-2 rounded-lg border text-slate-400 hover:text-white transition-colors flex items-center gap-1.5 text-xs font-mono ${showSettings ? 'bg-slate-800 border-slate-700 text-white' : 'bg-slate-900 border-slate-850'}`}
              title="Configurar Chave API"
            >
              <Settings className="w-4 h-4" />
              <span className="hidden sm:inline">Chave</span>
            </button>

            {/* BOTÃO DE SAÍDA / LOGOUT */}
            <button
              onClick={handleLogout}
              className="p-2 bg-red-950/20 hover:bg-red-950/40 border border-red-900 text-red-400 hover:text-red-300 rounded-lg text-xs font-mono transition-colors flex items-center gap-1"
              title="Encerrar Sessão"
            >
              <LogOut className="w-4 h-4" />
              <span className="hidden sm:inline">Sair</span>
            </button>
          </div>
        </div>
      </header>

      {/* PAINEL DE CONFIGURAÇÕES DE API */}
      {showSettings && (
        <div className="bg-slate-900 border-b border-slate-800 px-4 py-4 transition-all duration-200">
          <div className="max-w-3xl mx-auto bg-slate-950 p-4 rounded-xl border border-slate-800">
            <div className="flex justify-between items-start mb-2">
              <h3 className={`font-mono text-sm font-bold flex items-center gap-2 ${textAccent}`}>
                <Cpu className="w-4 h-4" />
                CONEXÃO COM OS SISTEMAS GOOGLE GEMINI
              </h3>
              <button onClick={() => setShowSettings(false)} className="text-xs text-slate-500 hover:text-slate-300">
                [ Fechar ]
              </button>
            </div>
            <p className="text-xs text-slate-400 mb-4 leading-relaxed">
              PRECHLAK está rodando com a sua chave ativa do Google AI Studio. Para configurar outra chave, use o campo:
            </p>
            <div className="flex flex-col sm:flex-row gap-2">
              <div className="relative flex-grow">
                <span className="absolute inset-y-0 left-0 flex items-center pl-3 text-slate-500">
                  <Lock className="w-3.5 h-3.5" />
                </span>
                <input 
                  type="password"
                  placeholder="Chave ativa carregada..."
                  value={customKey}
                  onChange={(e) => setCustomKey(e.target.value)}
                  className="w-full bg-slate-900 text-slate-100 pl-9 pr-3 py-2 text-xs rounded-lg border border-slate-800 focus:outline-none focus:border-slate-500 font-mono animate-pulse"
                />
              </div>
              <div className="flex gap-2">
                <button 
                  onClick={() => { setShowSettings(false); }}
                  className={`px-4 py-2 text-xs font-mono font-bold rounded-lg transition-all shadow-md w-full sm:w-auto ${buttonPrimary}`}
                >
                  Confirmar
                </button>
              </div>
            </div>
          </div>
        </div>
      )}

      {/* PAINEL PRINCIPAL */}
      <main className="flex-grow max-w-7xl w-full mx-auto p-4 flex flex-col lg:flex-row gap-4 overflow-hidden">
        
        {/* COLUNA ESQUERDA: NAVEGAÇÃO OPERACIONAL */}
        <nav className="w-full lg:w-64 flex flex-col gap-3 flex-shrink-0">
          
          <div className={`bg-slate-900/90 border rounded-xl p-3 flex flex-col shadow-xl gap-1 ${borderAccent}`}>
            <span className="text-[10px] font-mono text-slate-500 font-bold px-2 mb-1.5 uppercase tracking-widest">Atividade de Missão</span>
            
            <button
              onClick={() => setActiveTab('terminal')}
              className={`w-full flex items-center gap-2.5 px-3 py-2.5 rounded-lg text-xs font-mono font-bold transition-all text-left border ${
                activeTab === 'terminal' 
                  ? isMonochrome ? 'bg-slate-800 text-white border-slate-500 shadow-md' : 'bg-cyan-950/40 text-cyan-400 border-cyan-800/80' 
                  : 'bg-transparent text-slate-400 border-transparent hover:bg-slate-850 hover:text-slate-200'
              }`}
            >
              <Terminal className="w-4 h-4" />
              <span>Terminal de IA</span>
            </button>

            <button
              onClick={() => setActiveTab('osint')}
              className={`w-full flex items-center gap-2.5 px-3 py-2.5 rounded-lg text-xs font-mono font-bold transition-all text-left border ${
                activeTab === 'osint' 
                  ? isMonochrome ? 'bg-slate-800 text-white border-slate-500 shadow-md' : 'bg-cyan-950/40 text-cyan-400 border-cyan-800/80' 
                  : 'bg-transparent text-slate-400 border-transparent hover:bg-slate-850 hover:text-slate-200'
              }`}
            >
              <Globe className="w-4 h-4" />
              <span>Módulos OSINT</span>
            </button>

            <button
              onClick={() => setActiveTab('dossier')}
              className={`w-full flex items-center gap-2.5 px-3 py-2.5 rounded-lg text-xs font-mono font-bold transition-all text-left border ${
                activeTab === 'dossier' 
                  ? isMonochrome ? 'bg-slate-800 text-white border-slate-500 shadow-md' : 'bg-cyan-950/40 text-cyan-400 border-cyan-800/80' 
                  : 'bg-transparent text-slate-400 border-transparent hover:bg-slate-850 hover:text-slate-200'
              }`}
            >
              <FolderLock className="w-4 h-4" />
              <span>Central de Dossiês</span>
              {dossiers.length > 0 && (
                <span className={`ml-auto border text-[9px] px-1.5 py-0.2 rounded font-mono font-bold ${isMonochrome ? 'bg-slate-800 border-slate-700 text-slate-300' : 'bg-cyan-900/50 text-cyan-400 border-cyan-800'}`}>
                  {dossiers.length}
                </span>
              )}
            </button>

            {/* ABA DE ADMINISTRADOR (EXCLUSIVA DO USUÁRIO ADMIN) */}
            {currentUser?.isAdmin && (
              <button
                onClick={() => setActiveTab('admin')}
                className={`w-full flex items-center gap-2.5 px-3 py-2.5 rounded-lg text-xs font-mono font-bold transition-all text-left border ${
                  activeTab === 'admin' 
                    ? isMonochrome ? 'bg-slate-800 text-white border-slate-500 shadow-md' : 'bg-cyan-950/40 text-cyan-400 border-cyan-800/80' 
                    : 'bg-transparent text-slate-400 border-transparent hover:bg-slate-850 hover:text-slate-200'
                }`}
              >
                <UserPlus className="w-4 h-4 text-cyan-400" />
                <span>Painel Admin</span>
              </button>
            )}
          </div>

          {/* DOSSIÊS ATIVOS */}
          <div className={`bg-slate-900/90 border rounded-xl p-4 flex flex-col shadow-xl ${borderAccent}`}>
            <h3 className="text-[10px] font-bold font-mono tracking-widest text-slate-500 mb-3 uppercase flex items-center gap-1.5 border-b border-slate-850 pb-2">
              <Layers className={`w-3.5 h-3.5 ${textAccent}`} />
              Dossiês PRECHLAK ({dossiers.length})
            </h3>
            <div className="space-y-2 max-h-[180px] overflow-y-auto pr-1 custom-scrollbar font-mono">
              {dossiers.length === 0 ? (
                <span className="text-[10px] text-slate-500 font-mono block text-center py-2">Nenhum dossiê ativo</span>
              ) : (
                dossiers.map(d => (
                  <div key={d.id} className="p-1.5 bg-slate-950 rounded border border-slate-900 flex justify-between items-center gap-2">

                      <span className="text-[11px] text-slate-300 font-semibold truncate block mt-0.5">{d.alvo}</span>

                      d.risco === 'ALTO RISCO'

                        : isMonochrome ? 'border-slate-600 text-slate-300 bg-slate-950' : 'border-amber-800 text-amber-400 bg-amber-950/20'
                    }`}>
                      {d.risco.split(' ')[0]}

            <h3 className="text-[10px] font-bold font-mono tracking-widest text-slate-500 mb-2 uppercase flex items-center gap-1.5">
              <BookOpen className={`w-3.5 h-3.5 ${textAccent}`} />

            </h3>

                  key={t.id}

                  className="w-full text-left p-1.5 rounded bg-slate-950 hover:bg-slate-850 border border-slate-900 text-[11px] font-mono text-slate-400 hover:text-slate-200 transition-all flex items-center gap-1.5"
                >

                  <span>{t.title}</span>

        <section className={`flex-grow bg-slate-900/95 border rounded-xl flex flex-col shadow-2xl overflow-hidden min-h-[500px] ${borderAccent}`}>
          
          {/* TAB 1: TERMINAL DE IA CLASSICO */}
          {activeTab === 'terminal' && (
            <div className="flex flex-col flex-grow overflow-hidden h-[600px] lg:h-auto animate-in fade-in duration-150">
              
              <div className="bg-slate-950/90 px-4 py-2.5 border-b border-slate-800/90 flex justify-between items-center flex-shrink-0">

                    <span className={`relative inline-flex rounded-full h-2 w-2 ${isMonochrome ? 'bg-white' : 'bg-red-500'}`}></span>
                  </span>
                  <span className="font-mono text-xs font-bold text-slate-300">
                    CONSOLE TÁTICO PRECHLAK ACTIVATED

                <span className="text-[10px] font-mono text-slate-500 uppercase">
                  OPERADOR: {currentUser?.username}
                </span>
              </div>

              {/* Mensagens do chat */}
              <div className="flex-grow p-4 overflow-y-auto space-y-4 custom-scrollbar bg-[radial-gradient(ellipse_at_bottom_right,_var(--tw-gradient-stops))] from-slate-950/40 via-slate-900/90 to-slate-950/80">

          )}

          {/* TAB 3: CENTRAL DE DOSSIÊS (PRECHLAK COFRE) */}
          {activeTab === 'dossier' && (
            <div className="p-6 flex flex-col flex-grow overflow-y-auto space-y-6 custom-scrollbar animate-in fade-in duration-150">
              
              <div className="border-b border-slate-850 pb-3 flex flex-col md:flex-row md:items-center justify-between gap-3 font-mono">

                  <h2 className={`text-base font-bold flex items-center gap-2 ${textAccent}`}>
                    <FolderLock className="w-5 h-5" />

                <div className="flex gap-2 self-start md:self-auto font-mono">
                  <button

          {/* TAB 4: CONTROLE DE LICENÇAS ADMINISTRADOR (USUÁRIOS) */}
          {activeTab === 'admin' && currentUser?.isAdmin && (

                </p>
              </div>

              {/* Form de Criação de Novas Licenças */}
              <form onSubmit={handleCreateLogin} className="bg-slate-950 p-5 rounded-xl border border-slate-900 space-y-4 font-mono">
                <h3 className="text-xs font-bold uppercase text-cyan-400 flex items-center gap-1.5">
                  <UserPlus className="w-4 h-4" />
                  Gerar Credencial de Operador
                </h3>

                <div className="grid grid-cols-1 md:grid-cols-3 gap-3">
                  <div className="space-y-1">
                    <label className="text-[10px] font-bold text-slate-400 uppercase">Nome de Usuário:</label>
                    <input 
                      type="text"
                      placeholder="Ex: operador01"
                      value={newUsername}
                      onChange={(e) => setNewUsername(e.target.value)}
                      className="w-full bg-slate-900 border border-slate-800 rounded-lg p-2 text-xs text-slate-200 focus:outline-none focus:border-cyan-500"

                      type="text"
                      placeholder="Ex: pass9988"

                      onChange={(e) => setNewPassword(e.target.value)}

                    <select
                      value={activeDaysLimit}
                      onChange={(e) => setActiveDaysLimit(e.target.value)}

                      <option value="30">30 Dias (Mensal)</option>
                      <option value="90">90 Dias (Trimestral)</option>
                      <option value="365">365 Dias (Anual)</option>
                    </select>
                  </div>
                </div>

                <div className="flex justify-end pt-1">
                  <button type="submit" className={`px-4 py-2 font-bold text-xs rounded-lg transition-all ${buttonPrimary}`}>
                    Ativar Credencial
                  </button>
                </div>
              </form>

              {/* Lista de Usuários Ativos / Licenças */}
              <div className="space-y-3 font-mono">
                <h3 className="text-xs font-bold text-slate-400 uppercase flex items-center gap-1.5">
                  <Users className="w-4 h-4 text-slate-500" />
                  Operadores Cadastrados no Sistema
                </h3>

                <div className="bg-slate-950 rounded-xl border border-slate-900 overflow-hidden">

                    <thead>
                      <tr className="bg-slate-900/60 text-slate-400 border-b border-slate-850">
                        <th className="p-3">Usuário</th>
                        <th className="p-3">Senha</th>
                        <th className="p-3">Tipo</th>
                        <th className="p-3">Prazo</th>
                        <th className="p-3">Criação</th>
                        <th className="p-3 text-right">Ação</th>
                      </tr>
                    </thead>
                    <tbody>
                      {usersDb.map((user, idx) => {
                        const createdDate = new Date(user.createdAt);
                        const expirationDate = new Date(createdDate.getTime() + user.activeDays * 24 * 60 * 60 * 1000);
                        const isExpired = new Date() > expirationDate;

                        return (
                          <tr key={idx} className="border-b border-slate-900/60 hover:bg-slate-900/20">

                            <td className="p-3 text-slate-400">{user.password}</td>
                            <td className="p-3">
                              <span className={`px-1.5 py-0.5 rounded text-[10px] border ${user.isAdmin ? 'bg-cyan-950 text-cyan-400 border-cyan-800' : 'bg-slate-900 text-slate-400 border-slate-800'}`}>
                                {user.isAdmin ? 'ADMIN' : 'OPERADOR'}
                              </span>
                            </td>
                            <td className="p-3">
                              {user.isAdmin ? (
                                <span className="text-emerald-400 font-semibold">Ilimitado</span>

                                </span>
                              )}
                            </td>
                            <td className="p-3 text-slate-500">{createdDate.toLocaleDateString('pt-BR')}</td>
                            <td className="p-3 text-right">
                              {user.username !== 'prechlak' && (
                                <button
                                  onClick={() => handleDeleteUser(user.username)}
                                  className="text-red-500 hover:text-red-400 transition-colors"
                                  title="Remover Acesso"
                                >
                                  Remover
                                </button>
                              )}
                            </td>
                          </tr>
                        );
                      })}
                    </tbody>
                  </table>
                </div>
              </div>

            </div>
          )}

        </section>

      </main>

      {/* FOOTER */}
      <footer className="bg-slate-950 border-t border-slate-900 py-3 px-4 text-center text-xs text-slate-500 font-mono mt-auto flex-shrink-0">
        <div className="max-w-7xl mx-auto flex flex-col sm:flex-row justify-between items-center gap-2">
          <span>&copy; {new Date().getFullYear()} PRECHLAK. Licença autorizada de segurança e controle de acessos.</span>
          <span className={`text-[10px] flex items-center gap-1 ${textAccent}`}>
            <Shield className="w-3.5 h-3.5 animate-pulse" />
            CONEXÃO ATIVA NO CANAL DE CRIPTOGRAFIA ASSIMÉTRICA PRECHLAK SECURE
          </span>
        </div>
      </footer>
    </div>
  );
}
